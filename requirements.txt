# ==============================================================================
# Requisitos / Dependencias del proyecto Chess Game
# ==============================================================================
#
# Para instalar todas las dependencias necesarias en tu entorno virtual:
#     pip install -r requirements.txt
#
# Para ejecutar el juego tras la instalación:
#     python chess_game.py
#
# Requisitos del sistema:
#   - Python 3.8 o superior.
#   - Se recomienda el uso de un entorno virtual (venv):
#         python -m venv venv
#         source venv/bin/activate    # En Linux / macOS
#         venv\Scripts\activate       # En Windows
#
# IMPORTANTE PARA USUARIOS DE LINUX (Arch Linux, Ubuntu, Debian, Fedora):
# Si usas una versión muy reciente de Python (como 3.13 o 3.14), pip compila pygame
# desde el código fuente. Para que pygame incluya soporte para imágenes (JPEG/PNG)
# y tipografías (fuentes), debes tener instaladas las bibliotecas SDL2 del sistema:
#
#   - Arch Linux:
#         sudo pacman -S sdl2_image sdl2_ttf sdl2_mixer
#         pip install --no-cache-dir --force-reinstall pygame
#
#   - Ubuntu / Debian:
#         sudo apt install libsdl2-image-dev libsdl2-ttf-dev libsdl2-mixer-dev
#         pip install --no-cache-dir --force-reinstall pygame
#
#   - Alternativa directa (Pygame Community Edition con soporte moderno):
#         pip uninstall pygame
#         pip install pygame-ce
# ==============================================================================

# Pygame: Motor multimedia para la interfaz gráfica (GUI).
# Se encarga de:
#   - Renderizar el tablero, piezas, reloj, piezas capturadas y menús.
#   - Capturar eventos del ratón (selección de casillas, botones).
#   - Control de tasa de refresco (FPS) y bucle de juego.
pygame>=2.0.0

# python-chess: Biblioteca de reglas y lógica del ajedrez.
# Se encarga de:
#   - Representación del tablero e historial de movimientos.
#   - Generación y validación estricta de movimientos legales.
#   - Detección de jaque mate, tablas (ahogado, repetición, material insuficiente).
#   - Conversión de movimientos en notación UCI.
# Nota: Se importa en el código Python como 'import chess'.
python-chess>=1.9.0
